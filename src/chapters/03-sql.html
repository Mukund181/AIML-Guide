SELECT, WHERE & Filtering

SQL (Structured Query Language) is used to query relational databases. Data scientists use SQL daily to extract, transform, and analyze data. Think of a database as many Excel sheets (tables) that can be linked together.

SQL basics
-- Basic SELECT
SELECT name, age, salary
FROM employees
WHERE age > 30 AND salary > 50000
ORDER BY salary DESC
LIMIT 10;

-- LIKE (pattern matching)
SELECT * FROM customers
WHERE email LIKE '%@gmail.com';

-- IN, BETWEEN
SELECT * FROM orders
WHERE status IN ('shipped', 'delivered')
  AND order_date BETWEEN '2024-01-01' AND '2024-12-31';

-- CASE WHEN (like if-else)
SELECT name,
       salary,
       CASE
         WHEN salary > 100000 THEN 'Senior'
         WHEN salary > 60000  THEN 'Mid'
         ELSE 'Junior'
       END AS level
FROM employees;

Joins — Combining Tables

JOINs combine rows from two tables based on a related column. This is one of the most-tested SQL concepts in interviews.

Join TypeReturnsUse When
INNER JOINRows matching in BOTH tablesYou only want records that exist in both
LEFT JOINAll left rows + matching rightKeep all customers even if no orders
RIGHT JOINAll right rows + matching leftRarely used; flip tables and use LEFT JOIN
FULL OUTER JOINAll rows from both tablesWhen you need everything, matched or not
SELF JOINTable joined with itselfEmployee → Manager hierarchy
SQL joins
-- INNER JOIN: customers who placed orders
SELECT c.name, o.order_id, o.amount
FROM customers c
INNER JOIN orders o ON c.customer_id = o.customer_id;

-- LEFT JOIN: all customers, even those with NO orders (NULL)
SELECT c.name, o.order_id
FROM customers c
LEFT JOIN orders o ON c.customer_id = o.customer_id;

-- Find customers with NO orders (anti-join pattern)
SELECT c.name
FROM customers c
LEFT JOIN orders o ON c.customer_id = o.customer_id
WHERE o.order_id IS NULL;

-- SELF JOIN: employee → manager
SELECT e.name AS employee, m.name AS manager
FROM employees e
LEFT JOIN employees m ON e.manager_id = m.employee_id;

Aggregations & GROUP BY

Aggregation functions compute summary statistics over groups of rows.

SQL aggregations
-- Aggregate functions: COUNT, SUM, AVG, MAX, MIN
SELECT
  department,
  COUNT(*) AS num_employees,
  AVG(salary) AS avg_salary,
  MAX(salary) AS top_salary,
  SUM(salary) AS total_payroll
FROM employees
GROUP BY department
HAVING COUNT(*) > 5          -- HAVING filters AFTER grouping (WHERE is before)
ORDER BY avg_salary DESC;

-- COUNT DISTINCT
SELECT COUNT(DISTINCT customer_id) AS unique_customers
FROM orders
WHERE order_date >= '2024-01-01';
📌 WHERE vs HAVING WHERE filters rows BEFORE grouping. HAVING filters groups AFTER GROUP BY. If you want "departments with more than 5 people", use HAVING COUNT(*) > 5, not WHERE.

CTEs & Subqueries

CTEs (Common Table Expressions) make complex queries readable by naming intermediate results. Subqueries are nested SELECT statements.

CTEs and subqueries
-- Subquery: top-earning employees per department
SELECT name, salary, department
FROM employees
WHERE salary = (
  SELECT MAX(salary) FROM employees e2
  WHERE e2.department = employees.department
);

-- CTE: same query — much more readable!
WITH dept_max AS (
  SELECT department, MAX(salary) AS max_salary
  FROM employees
  GROUP BY department
)
SELECT e.name, e.salary, e.department
FROM employees e
JOIN dept_max d ON e.department = d.department
                AND e.salary = d.max_salary;

-- Chained CTEs
WITH
  active_users AS (
    SELECT user_id FROM users WHERE status = 'active'
  ),
  user_orders AS (
    SELECT o.user_id, SUM(o.amount) AS total
    FROM orders o
    JOIN active_users a ON o.user_id = a.user_id
    GROUP BY o.user_id
  )
SELECT * FROM user_orders WHERE total > 1000;

Window Functions

Window functions perform calculations across a set of rows related to the current row — without collapsing them into groups. This is a high-signal interview topic.

window functions
-- OVER() = "across these rows"
-- Syntax: function() OVER (PARTITION BY col ORDER BY col)

-- ROW_NUMBER: rank employees within each department by salary
SELECT
  name, department, salary,
  ROW_NUMBER() OVER (PARTITION BY department ORDER BY salary DESC) AS rn
FROM employees;

-- Get the TOP 1 salary per department using ROW_NUMBER
WITH ranked AS (
  SELECT name, department, salary,
    ROW_NUMBER() OVER (PARTITION BY department ORDER BY salary DESC) AS rn
  FROM employees
)
SELECT * FROM ranked WHERE rn = 1;

-- RANK vs DENSE_RANK (ties matter!)
-- RANK:       1, 2, 2, 4  (skips 3 after tie)
-- DENSE_RANK: 1, 2, 2, 3  (no gaps)
SELECT name, salary,
  RANK()       OVER (ORDER BY salary DESC) AS rnk,
  DENSE_RANK() OVER (ORDER BY salary DESC) AS dense_rnk
FROM employees;

-- Running total (cumulative sum)
SELECT order_date, amount,
  SUM(amount) OVER (ORDER BY order_date) AS running_total
FROM orders;

-- LAG / LEAD: compare with previous / next row
SELECT order_date, amount,
  LAG(amount)  OVER (ORDER BY order_date) AS prev_amount,
  LEAD(amount) OVER (ORDER BY order_date) AS next_amount,
  amount - LAG(amount) OVER (ORDER BY order_date) AS day_over_day_change
FROM orders;
FunctionWhat it does
ROW_NUMBER()Unique sequential rank (1,2,3,4) — no ties
RANK()Rank with gaps on ties (1,2,2,4)
DENSE_RANK()Rank without gaps on ties (1,2,2,3)
SUM() OVER()Running/cumulative total
LAG(col, n)Value from n rows before
LEAD(col, n)Value from n rows ahead
NTILE(n)Divides rows into n equal buckets

Indexes & Query Optimization

Indexes speed up queries by creating a fast lookup structure (like a book's index). Without them, the database scans every row.

indexes and optimization
-- Create an index on a frequently filtered column
CREATE INDEX idx_emp_dept ON employees(department);
CREATE INDEX idx_orders_date ON orders(order_date);

-- Composite index (for queries filtering by both columns)
CREATE INDEX idx_emp_dept_salary ON employees(department, salary);

-- EXPLAIN: see how the database runs your query
EXPLAIN SELECT * FROM employees WHERE department = 'Engineering';
-- Look for: "Index Scan" (fast) vs "Seq Scan" (slow full table scan)
⚡

Use Indexes On

Columns in WHERE, JOIN ON, and ORDER BY clauses. Foreign keys. High-cardinality columns (many unique values like user_id).

🐢

Common Slow Query Patterns

SELECT * (fetch only needed columns), LIKE '%word%' (can't use index), functions on indexed columns: WHERE YEAR(date)=2024 → use WHERE date >= '2024-01-01'.

🔧

Optimization Tips

Filter early with WHERE. Use JOINs instead of correlated subqueries. Avoid SELECT * in production. Use LIMIT when exploring. Analyze with EXPLAIN.

import sqlite3
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# ==============================================================================
# STEP 1: INITIALIZE SQLITE DATABASE & SEED REAL-WORLD RELATIONAL DATA
# ==============================================================================
conn = sqlite3.connect(":memory:")  # In-memory database for reproducible execution
cur = conn.cursor()

# 1. Create Relational Tables
cur.executescript("""
CREATE TABLE customers (
    customer_id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    country TEXT NOT NULL,
    signup_date DATE NOT NULL
);

CREATE TABLE products (
    product_id INTEGER PRIMARY KEY,
    product_name TEXT NOT NULL,
    category TEXT NOT NULL,
    price REAL NOT NULL
);

CREATE TABLE orders (
    order_id INTEGER PRIMARY KEY,
    customer_id INTEGER NOT NULL,
    order_date DATE NOT NULL,
    status TEXT NOT NULL,
    FOREIGN KEY (customer_id) REFERENCES customers(customer_id)
);

CREATE TABLE order_items (
    item_id INTEGER PRIMARY KEY,
    order_id INTEGER NOT NULL,
    product_id INTEGER NOT NULL,
    quantity INTEGER NOT NULL,
    FOREIGN KEY (order_id) REFERENCES orders(order_id),
    FOREIGN KEY (product_id) REFERENCES products(product_id)
);
""")

# 2. Seed Realistic Data
cur.executemany("INSERT INTO customers VALUES (?, ?, ?, ?)", [

cur.executemany("INSERT INTO products VALUES (?, ?, ?, ?)", [

cur.executemany("INSERT INTO orders VALUES (?, ?, ?, ?)", [
    (1001, 1, '2023-01-20', 'delivered'),
    (1002, 2, '2023-02-14', 'delivered'),
    (1003, 1, '2023-03-10', 'delivered'),
    (1004, 3, '2023-03-15', 'delivered'),

cur.executemany("INSERT INTO order_items VALUES (?, ?, ?, ?)", [
    (1, 1001, 101, 1),  # Aarav: Laptop ($1899.99)
    (2, 1001, 104, 1),  # Aarav: Mouse ($79.99)
    (3, 1002, 103, 1),  # Elena: Monitor ($649.00)

    (10,1009, 104, 1),  # Priya: Mouse ($79.99)

])
conn.commit()

# ==============================================================================
# STEP 2: ADVANCED SQL ANALYTICAL QUERY (CTEs + MULTI-JOINS + WINDOW FUNCTIONS)
# ==============================================================================
analytical_query = """
WITH order_values AS (
    -- 1. Roll up item-level transactions into total order spend
    SELECT 
        o.order_id,
        o.customer_id,
        o.order_date,
        ROUND(SUM(oi.quantity * p.price), 2) AS order_revenue,
        SUM(oi.quantity) AS total_items
    FROM orders o
    JOIN order_items oi ON o.order_id = oi.order_id
    JOIN products p     ON oi.product_id = p.product_id
    GROUP BY o.order_id, o.customer_id, o.order_date
),
customer_lifecycle AS (
    -- 2. Compute customer repeat behavior using Window Functions
    SELECT 
        c.customer_id,
        c.name,
        c.country,
        ov.order_id,
        ov.order_date,
        ov.order_revenue,
        ROW_NUMBER() OVER (
            PARTITION BY c.customer_id ORDER BY ov.order_date ASC
        ) AS purchase_sequence,
        LAG(ov.order_date) OVER (
            PARTITION BY c.customer_id ORDER BY ov.order_date ASC
        ) AS prev_order_date,
        SUM(ov.order_revenue) OVER (
            ORDER BY ov.order_date ASC
        ) AS cumulative_running_revenue
    FROM customers c
    JOIN order_values ov ON c.customer_id = ov.customer_id
)
SELECT 
    customer_id,
    name,
    country,
    order_id,
    order_date,
    order_revenue,
    purchase_sequence,
    prev_order_date,
    ROUND(cumulative_running_revenue, 2) AS running_revenue,
    DENSE_RANK() OVER (ORDER BY order_revenue DESC) AS order_revenue_rank
FROM customer_lifecycle
ORDER BY order_date ASC;
"""

# ==============================================================================
# STEP 3: DIRECT INGESTION INTO PANDAS & STATISTICAL COMPUTATION
# ==============================================================================
df_lifecycle = pd.read_sql_query(analytical_query, conn)

# Parse datetime for intervals
df_lifecycle['order_date'] = pd.to_datetime(df_lifecycle['order_date'])
df_lifecycle['prev_order_date'] = pd.to_datetime(df_lifecycle['prev_order_date'])
df_lifecycle['days_since_last_order'] = (
    df_lifecycle['order_date'] - df_lifecycle['prev_order_date']
).dt.days

# Customer Lifetime Value (CLV) rollup
df_clv = df_lifecycle.groupby(['customer_id', 'name', 'country']).agg(
    total_spent=('order_revenue', 'sum'),
    total_orders=('order_id', 'count'),
    avg_order_value=('order_revenue', 'mean')
).reset_index()

df_clv['clv_rank'] = df_clv['total_spent'].rank(ascending=False, method='dense')

# Query category performance for business intelligence
category_query = """
SELECT 
    p.category,
    SUM(oi.quantity) AS total_units_sold,
    ROUND(SUM(oi.quantity * p.price), 2) AS gross_revenue
FROM products p
JOIN order_items oi ON p.product_id = oi.product_id
GROUP BY p.category
ORDER BY gross_revenue DESC;
"""
df_categories = pd.read_sql_query(category_query, conn)

# Output Business Insights
print("=== CUSTOMER LIFETIME VALUE (CLV) LEADERBOARD ===")
print(df_clv.sort_values('total_spent', ascending=False).to_string(index=False))

repeat_customers = df_clv[df_clv['total_orders'] > 1]
avg_repurchase_days = df_lifecycle['days_since_last_order'].mean()

print(f"\nTotal Revenue: ${df_lifecycle['order_revenue'].sum():,.2f}")
print(f"Repeat Customer Rate: {len(repeat_customers)/len(df_clv):.1%}")
print(f"Mean Repurchase Cadence: {avg_repurchase_days:.1f} days")

# ==============================================================================
# STEP 4: VISUAL STORYTELLING (MATPLOTLIB)
# ==============================================================================
fig, axes = plt.subplots(2, 2, figsize=(12, 8))
plt.subplots_adjust(hspace=0.35, wspace=0.25)

# Plot 1: Cumulative Revenue Growth
axes[0, 0].plot(df_lifecycle['order_date'], df_lifecycle['running_revenue'], 
                 marker='o', color='#20c997', linewidth=2.5)
axes[0, 0].set_title('Cumulative Revenue Over Time', fontweight='bold')
axes[0, 0].set_ylabel('Revenue ($)')
axes[0, 0].grid(True, linestyle='--', alpha=0.5)

# Plot 2: Total Spent by Customer
axes[0, 1].barh(df_clv['name'], df_clv['total_spent'], color='#339af0')
axes[0, 1].set_title('Customer Lifetime Value (CLV)', fontweight='bold')
axes[0, 1].set_xlabel('Total Spend ($)')

# Plot 3: Revenue by Category
axes[1, 0].pie(df_categories['gross_revenue'], labels=df_categories['category'], 
                 autopct='%1.1f%%', colors=['#ff6b6b', '#51cf66', '#fcc419'])
axes[1, 0].set_title('Gross Revenue Share by Category', fontweight='bold')

# Plot 4: Repurchase Interval Distribution
valid_intervals = df_lifecycle['days_since_last_order'].dropna()
axes[1, 1].hist(valid_intervals, bins=5, color='#cc5de8', edgecolor='black')
axes[1, 1].set_title('Repurchase Cadence Distribution', fontweight='bold')
axes[1, 1].set_xlabel('Days Between Orders')
axes[1, 1].set_ylabel('Number of Repeat Orders')

plt.suptitle('Phase 1 Deliverable: E-Commerce Analytics Dashboard', fontsize=14, fontweight='bold')
plt.show()
